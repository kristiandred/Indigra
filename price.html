function loadPricesFromStorage() { console.log('Загружаем цены из хранилища...'); // Проверяем оба возможных ключа от админки const keys = ['price_data_tarif.xlsx', 'price_data_Nasos.xlsx']; let allChanges = {}; keys.forEach(key => { const savedData = localStorage.getItem(key); if (savedData) { try { const changes = JSON.parse(savedData); allChanges = {...allChanges, ...changes}; console.log(`✅ Загружены данные из ${key}:`, Object.keys(changes).length, 'изменений'); } catch (e) { console.error(`❌ Ошибка загрузки из ${key}:`, e); } } }); if (Object.keys(allChanges).length === 0) { console.log('ℹ️ Нет сохраненных изменений'); return; } console.log('📊 Все изменения:', allChanges); // Применяем изменения к таблицам let updatedCount = 0; for (const [cellId, value] of Object.entries(allChanges)) { // ID ячейки выглядит как "table-abs-main_0_1" или "table-abs-main_0" const parts = cellId.split('_'); // Определяем ID таблицы (без последнего индекса, если он есть) let tableId; let rowIndex; let colIndex = 0; if (parts.length === 2) { // Формат: "table-id_номер_строки" tableId = parts[0]; rowIndex = parseInt(parts[1]); } else if (parts.length >= 3) { // Формат: "table-id_0_номер_строки" или "table-id_0_номер_строки_номер_колонки" // Пропускаем средний индекс (обычно 0) и берем таблицу без него const tableParts = parts.slice(0, -2); // Убираем два последних элемента tableId = tableParts.join('_'); rowIndex = parseInt(parts[parts.length - 2]); // Предпоследний - номер строки colIndex = parseInt(parts[parts.length - 1]); // Последний - номер колонки } else { console.log(`❌ Неверный формат ID: ${cellId}`); continue; } console.log(`🔍 Ищем: таблица=${tableId}, строка=${rowIndex}, колонка=${colIndex}, значение=${value}`); const table = document.getElementById(tableId); if (table) { const tbody = table.querySelector('tbody'); if (tbody) { const rows = tbody.querySelectorAll('tr'); if (rows[rowIndex]) { const cells = rows[rowIndex].querySelectorAll('td'); if (cells[colIndex]) { // Ячейка найдена, обновляем значение const cell = cells[colIndex]; const priceSpan = cell.querySelector('.badge-price'); if (priceSpan) { priceSpan.textContent = value; console.log(`✅ Обновлена цена в ${tableId}: ${value}`); } else { cell.textContent = value; console.log(`✅ Обновлен текст в ${tableId}: ${value}`); } updatedCount++; } else { console.log(`❌ Нет колонки ${colIndex} в строке ${rowIndex}`); } } else { console.log(`❌ Нет строки ${rowIndex} в таблице ${tableId}`); } } else { console.log(`❌ В таблице ${tableId} нет tbody`); } } else { console.log(`❌ Таблица ${tableId} не найдена`); } } console.log(`🎉 Всего обновлено ${updatedCount} ячеек`); }
